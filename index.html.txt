    <script>
        // 数据存储
        let studentList = JSON.parse(localStorage.getItem("innovRollCall_students")) || [];
        let callRecord = JSON.parse(localStorage.getItem("innovRollCall_record")) || [];
        let config = JSON.parse(localStorage.getItem("innovRollCall_config")) || {theme:"blackboard"};
        let currentSelectStudent = null;
        let rollTimer = null;
        let heatChart = null;

        // 🆕 签到相关变量
        let checkinTimer = null;
        let qrRefreshTimer = null;
        let checkinRemaining = 0;
        let qrcodeObj = null;

        // 鼓励评语库
        const encourageWords = [
            "很棒，期待你的精彩回答！",
            "相信你一定有独到见解👍",
            "大胆说出你的想法，课堂不怕犯错",
            "这是一次很好的锻炼机会",
            "准备好了吗？全班同学都在倾听"
        ];

        // 初始化
        function init(){
            if(config.theme === "tech"){
                document.getElementById("appBody").classList.add("tech-bg");
            }else{
                document.getElementById("appBody").classList.add("blackboard-bg");
            }
            renderListDom();
            renderRecordDom();
            renderHeatChart();
            updateStatPanel();
            document.getElementById("weightMode").checked = config.weightMode ?? true;
            document.getElementById("groupMode").checked = config.groupMode ?? false;
            document.getElementById("handUpMode").checked = config.handUpMode ?? false;
            // 清空二维码初始占位
            document.getElementById("qrcode").innerHTML = "";
        }

        // 切换主题
        function toggleTheme(){
            const body = document.getElementById("appBody");
            body.classList.remove("blackboard-bg","tech-bg");
            if(config.theme === "blackboard"){
                config.theme = "tech";
                body.classList.add("tech-bg");
            }else{
                config.theme = "blackboard";
                body.classList.add("blackboard-bg");
            }
            saveConfig();
        }

        function saveConfig(){
            config.weightMode = document.getElementById("weightMode").checked;
            config.groupMode = document.getElementById("groupMode").checked;
            config.handUpMode = document.getElementById("handUpMode").checked;
            localStorage.setItem("innovRollCall_config", JSON.stringify(config));
        }

        // 加权随机算法【核心创新】
        function weightedPick(){
            if(studentList.length ===0) return null;
            const weightArr = studentList.map(item=>{
                const baseWeight = Math.max(1, 10 - (item.callCount||0));
                return {...item, w: baseWeight};
            })
            let totalW = weightArr.reduce((sum,i)=>sum+i.w,0);
            let rand = Math.random() * totalW;
            for(let s of weightArr){
                rand -= s.w;
                if(rand <=0) return s;
            }
            return weightArr[0];
        }

        // 普通随机
        function normalPick(){
            return studentList[Math.floor(Math.random()*studentList.length)];
        }

        // 开始滚动点名
        function startRoll(){
            if(studentList.length === 0){
                alert("请先导入学生名单！");
                return;
            }
            document.getElementById("btnStart").disabled = true;
            document.getElementById("btnStop").disabled = false;
            document.getElementById("resultAction").classList.add("hidden");
            const nameBox = document.getElementById("nameBox");
            rollTimer = setInterval(()=>{
                const pick = config.weightMode ? weightedPick() : normalPick();
                nameBox.innerText = pick.name;
                nameBox.classList.add("roll-animate");
            },80)
        }

        function stopRoll(){
            clearInterval(rollTimer);
            document.getElementById("btnStart").disabled = false;
            document.getElementById("btnStop").disabled = true;
            const pick = config.weightMode ? weightedPick() : normalPick();
            currentSelectStudent = pick;
            document.getElementById("nameBox").innerText = pick.name;
            document.getElementById("nameBox").classList.remove("roll-animate");
            document.getElementById("resultAction").classList.remove("hidden");
            document.getElementById("encourageWord").innerText = encourageWords[Math.floor(Math.random()*encourageWords.length)];
            fireworkAnim();
        }

        // 设置出勤状态
        function setStatus(status){
            if(!currentSelectStudent) return;
            const idx = studentList.findIndex(x=>x.name === currentSelectStudent.name);
            if(idx !== -1){
                studentList[idx].callCount = (studentList[idx].callCount||0)+1;
            }
            const rec = {
                name: currentSelectStudent.name,
                status,
                time: new Date().toLocaleString()
            }
            callRecord.push(rec);
            localStorage.setItem("innovRollCall_students", JSON.stringify(studentList));
            localStorage.setItem("innovRollCall_record", JSON.stringify(callRecord));
            currentSelectStudent = null;
            document.getElementById("resultAction").classList.add("hidden");
            document.getElementById("nameBox").innerText = "准备开始智能点名";
            renderListDom();
            renderRecordDom();
            renderHeatChart();
            updateStatPanel();
        }

        // 📊 更新考勤统计面板
        function updateStatPanel(){
            const total = studentList.length;
            const present = callRecord.filter(r=>r.status === "present").length;
            const late = callRecord.filter(r=>r.status === "late").length;
            const absent = callRecord.filter(r=>r.status === "absent").length;
            const checked = present + late + absent;
            const rate = checked>0 ? Math.round( (present / checked)*100 ) : 0;

            document.getElementById("statTotal").innerText = total;
            document.getElementById("statChecked").innerText = checked;
            document.getElementById("statPresent").innerText = present;
            document.getElementById("statLate").innerText = late;
            document.getElementById("statAbsent").innerText = absent;
            document.getElementById("statRate").innerText = rate + "%";
        }

        // 🆕 签到模块：生成动态二维码（随机密钥）
        function refreshQrCode(){
            document.getElementById("qrcode").innerHTML = "";
            // 动态随机密钥，每次刷新二维码内容变化
            const secretKey = "checkin_" + Date.now() + "_" + Math.random().toString(36).slice(2,8);
            qrcodeObj = new QRCode(document.getElementById("qrcode"), {
                text: secretKey,
                width:140,
                height:140
            })
        }

        // 🆕 发起签到
        function startCheckIn(){
            const inputVal = Number(document.getElementById("checkinTimeInput").value);
            if(isNaN(inputVal) || inputVal <10 || inputVal>300){
                alert("签到时长范围：10~300秒");
                return;
            }
            stopCheckIn();
            checkinRemaining = inputVal;
            refreshQrCode();
            // 每5秒刷新一次二维码，动态密钥，防截图代签
            qrRefreshTimer = setInterval(()=>{
                refreshQrCode();
            },5000)

            checkinTimer = setInterval(()=>{
                checkinRemaining--;
                document.getElementById("countdownText").innerText = `剩余 ${checkinRemaining} s`;
                if(checkinRemaining <= 0){
                    stopCheckIn();
                }
            },1000)
        }

        // 🆕 结束签到
        function stopCheckIn(){
            clearInterval(checkinTimer);
            clearInterval(qrRefreshTimer);
            checkinRemaining = 0;
            document.getElementById("countdownText").innerText = "签到已结束";
        }

        // 导入名单
        function loadStudentList(){
            const text = document.getElementById("studentInput").value.trim();
            if(!text) return alert("请输入名单");
            const raw = text.split(/[\n,，]/).map(i=>i.trim()).filter(i=>i);
            let newList = [];
            raw.forEach(name=>{
                const exist = studentList.find(s=>s.name === name);
                if(exist){
                    newList.push(exist);
                }else{
                    newList.push({name, callCount:0, tag:"normal"})
                }
            })
            studentList = newList;
            localStorage.setItem("innovRollCall_students", JSON.stringify(studentList));
            renderListDom();
            renderHeatChart();
            updateStatPanel();
            alert(`成功导入 ${studentList.length} 位同学`);
        }

        function clearAllList(){
            if(!confirm("确定清空名单与记录？")) return;
            studentList = [];
            callRecord = [];
            localStorage.setItem("innovRollCall_students", JSON.stringify(studentList));
            localStorage.setItem("innovRollCall_record", JSON.stringify(callRecord));
            renderListDom();
            renderRecordDom();
            renderHeatChart();
            updateStatPanel();
        }

        function renderListDom(){
            const dom = document.getElementById("nameListDom");
            dom.innerHTML = studentList.map(i=>`
                <div class="flex justify-between py-1 border-b border-white/10">
                    <span>${i.name}</span>
                    <span class="opacity-60">点名${i.callCount||0}次</span>
                </div>
            `).join("")
        }

        function renderRecordDom(){
            const dom = document.getElementById("recordDom");
            dom.innerHTML = callRecord.slice(-12).reverse().map(r=>{
                let statusText = r.status;
                if(r.status==="present") statusText="✅到课";
                if(r.status==="late") statusText="⏱️迟到";
                if(r.status==="absent") statusText="❌缺勤";
                return `
                <div class="py-1 border-b border-white/10 text-xs">
                    ${r.time}｜${r.name}：${statusText}
                </div>
                `
            }).join("") || "暂无点名记录"
        }

        // 热力图表
        function renderHeatChart(){
            const ctx = document.getElementById("heatChart").getContext("2d");
            const labels = studentList.map(i=>i.name);
            const data = studentList.map(i=>i.callCount||0);
            if(heatChart) heatChart.destroy();
            heatChart = new Chart(ctx,{
                type:"bar",
                data:{
                    labels,
                    datasets:[{
                        label:"点名次数",
                        data,
                        backgroundColor:"rgba(250,220,80,0.6)"
                    }]
                },
                options:{
                    responsive:true,
                    scales:{
                        y:{beginAtZero:true,ticks:{color:"#f8f5e6"}},
                        x:{ticks:{color:"#f8f5e6",maxRotation:45,minRotation:0}}
                    },
                    plugins:{legend:{labels:{color:"#f8f5e6"}}}
                }
            })
        }

        // 礼花特效
        function fireworkAnim(){
            const canvas = document.getElementById("fireworkCanvas");
            const ctx = canvas.getContext("2d");
            canvas.width = canvas.clientWidth;
            canvas.height = canvas.clientHeight;
            let particles = [];
            for(let i=0;i<60;i++){
                const angle = Math.PI*2/60*i;
                particles.push({
                    x:canvas.width/2,
                    y:canvas.height/2,
                    vx:Math.cos(angle)*3,
                    vy:Math.sin(angle)*3,
                    life:1
                })
            }
            function draw(){
                ctx.clearRect(0,0,canvas.width,canvas.height);
                particles.forEach(p=>{
                    p.x += p.vx;
                    p.y += p.vy;
                    p.life -= 0.02;
                    ctx.beginPath();
                    ctx.arc(p.x,p.y,4,0,Math.PI*2);
                    ctx.fillStyle = `rgba(255,220,80,${p.life})`;
                    ctx.fill();
                })
                if(particles[0].life >0) requestAnimationFrame(draw);
            }
            draw();
        }

        init();
